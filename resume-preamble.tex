% Base preamble for ATS Safe Resume
% Included via --include-in-header, after the theme file
%
% Accent bars under section headings (uses {accent} color from theme)
\addtokomafont{section}{\color{accent}\large}
\addtokomafont{subsection}{\color{accent}\normalsize}
\let\oldsection\section
\renewcommand{\section}[1]{\oldsection{#1}\vspace{-0.5em}\par\noindent\textcolor{accent!40}{\rule{\textwidth}{0.3pt}}\vspace{0.3em}}

% Compact list and paragraph spacing (enumitem replaces deprecated paralist)
\usepackage{enumitem}
\setlist{nosep,leftmargin=*}
\setlength{\parskip}{0.4\baselineskip}
\setlength{\parsep}{0pt}
\setlength{\itemsep}{0pt}
\setlength{\topsep}{0pt}
\setlength{\partopsep}{0pt}
